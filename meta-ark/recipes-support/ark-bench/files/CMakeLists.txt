cmake_minimum_required(VERSION 3.21)
project(ark_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ARK_ENABLE_CUDA "Build the target CUDA correctness/throughput probe" ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(ark-cpu-bench cpu-bench.cpp)
target_link_libraries(ark-cpu-bench PRIVATE OpenSSL::Crypto)
install(TARGETS ark-cpu-bench RUNTIME DESTINATION bin)
if(ARK_ENABLE_CUDA)
    enable_language(CUDA)
    add_executable(ark-cuda-bench cuda-bench.cu)
    set_target_properties(ark-cuda-bench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Shared)
    install(TARGETS ark-cuda-bench RUNTIME DESTINATION bin)
endif()
